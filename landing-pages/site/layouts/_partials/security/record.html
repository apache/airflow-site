$cfg := hugo.Data.security.products -}} {{- $meta := $rec.cveMetadata -}} {{- $cna := $rec.containers.cna -}} {{- /* English description, else the first one. */ -}} {{- $desc := "" -}} {{- range $cna.descriptions -}} {{- if and (not $desc) (hasPrefix (lower (.lang | default "")) "en") -}}{{- $desc = .value -}}{{- end -}} {{- end -}} {{- if and (not $desc) $cna.descriptions -}}{{- $desc = (index $cna.descriptions 0).value -}}{{- end -}} {{- $desc = trim (replace (string $desc) "\r\n" "\n") " \t\n" -}} {{- /* Older records have no title: use the start of the description. */ -}} {{- $title := trim (string ($cna.title | default "")) " \t\n" -}} {{- if not $title -}} {{- $title = replaceRE `\s+` " " $desc -}} {{- if gt (len $title) 100 -}}{{- $title = printf "%s…" (trim (substr $title 0 100) " ") -}}{{- end -}} {{- end -}} {{- /* The ASF severity rating is carried as free text in metrics[].other. */ -}} {{- $severity := "" -}} {{- range $cna.metrics -}} {{- with .other -}}{{- with .content -}} {{- if not $severity -}}{{- $severity = trim (string (.text | default .other | default "")) " " -}}{{- end -}} {{- end -}}{{- end -}} {{- end -}} {{- /* One entry per affected[] item, assigned to a product listing. */ -}} {{- $entries := slice -}} {{- $categories := slice -}} {{- $labels := slice -}} {{- $packages := slice -}} {{- range $cna.affected -}} {{- $name := trim (string (.product | default "")) " " -}} {{- $pkg := replaceRE `^([a-z0-9._-]*).*$` "$1" (lower (string (.packageName | default ""))) -}} {{- if not $pkg -}}{{- $pkg = lower (string (index $cfg.aliases $name | default "")) -}}{{- end -}} {{- $cat := "" -}} {{- $catTitle := "" -}} {{- range $cfg.products -}} {{- if not $cat -}} {{- if or (and $pkg (in (.packages | default slice) $pkg)) (and $pkg .package_prefix (hasPrefix $pkg .package_prefix)) (in (.names | default slice) $name) -}} {{- $cat = .key -}}{{- $catTitle = .title -}} {{- end -}} {{- end -}} {{- end -}} {{- $label := $catTitle -}} {{- if not $cat -}}{{- $cat = "other" -}}{{- $label = $name -}}{{- end -}} {{- if eq $cat "providers" -}}{{- $label = $pkg -}}{{- $packages = $packages | append $pkg -}}{{- end -}} {{- /* Ranges for the version filter, as {from, lt, le} (dotted versions or ""): `version` is the lower bound ("0" = none), with `lessThan` or `lessThanOrEqual` as the upper one, or on its own a single version. */ -}} {{- $ranges := slice -}} {{- $isVersion := `^\d+(?:\.\d+)*$` -}} {{- range (.versions | default slice) -}} {{- if and .status (ne .status "affected") -}}{{- continue -}}{{- end -}} {{- $v := trim (string (.version | default "")) " " -}} {{- $lt := trim (string (.lessThan | default "")) " " -}} {{- $le := trim (string (.lessThanOrEqual | default "")) " " -}} {{- $from := cond (findRE $isVersion $v) $v "" -}} {{- $upper := or $lt $le -}} {{- if and $upper (not (findRE $isVersion $upper)) -}} {{- warnf "%s: upper bound %q is not a version number, the version filter ignores it" $meta.cveId $upper -}} {{- else if $lt -}} {{- $ranges = $ranges | append (dict "from" $from "lt" $lt "le" "") -}} {{- else if $le -}} {{- $ranges = $ranges | append (dict "from" $from "lt" "" "le" $le) -}} {{- else if $from -}} {{- $ranges = $ranges | append (dict "from" $from "lt" "" "le" $from) -}} {{- else -}} {{- warnf "%s: affected version %q is not a version number, the version filter ignores it" $meta.cveId $v -}} {{- end -}} {{- end -}} {{- $entries = $entries | append (dict "product" $name "package" $pkg "category" $cat "label" $label "versions" (.versions | default slice) "ranges" $ranges "default_status" (.defaultStatus | default "")) -}} {{- $categories = $categories | append $cat -}} {{- $labels = $labels | append $label -}} {{- end -}} {{- $published := time.AsTime $meta.datePublished -}} {{- $updated := $published -}} {{- with $meta.dateUpdated -}}{{- $updated = time.AsTime . -}}{{- end -}} {{- return (dict "id" $meta.cveId "state" $meta.state "title" $title "description" $desc "severity" $severity "published" $published "updated" $updated "entries" $entries "categories" (uniq $categories) "labels" (uniq $labels) "packages" (uniq $packages) "problem_types" ($cna.problemTypes | default slice) "credits" ($cna.credits | default slice) "references" ($cna.references | default slice) ) -}}
